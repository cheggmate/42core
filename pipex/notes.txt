// create a pipe
int pipe_fd[2];
if (pipe(pipe_fd) == -1) {
    // error handling
}
// pipe_fd[0] is read end, pipe_fd[1] is write end

// first child process - handles file input
pid1 = fork();
if (pid1 == 0) { // Child process 1
    close(pipe_fd[0]); // Close read end, we won't use it
    
    // Redirect stdin to read from the file
    int input_fd = open(argv[1], O_RDONLY);
    if (input_fd == -1) {
        // error handling
    }
    dup2(input_fd, STDIN_FILENO); // Now stdin comes from the file
    close(input_fd);
    
    // Redirect stdout to write to the pipe
    dup2(pipe_fd[1], STDOUT_FILENO);
    close(pipe_fd[1]);
    2
    // Execute the first command
    char **cmd1_args = parse_command(argv[2]);
    execve(cmd1_args[0], cmd1_args, environ);
}

// second child process - handles command processing
pid2 = fork();
if (pid2 == 0) { // Child process 2
    close(pipe_fd[1]); // Close write end, we won't use it
    
    // Redirect stdin to read from the pipe
    dup2(pipe_fd[0], STDIN_FILENO);
    close(pipe_fd[0]);
    
    // Redirect stdout to write to output file
    int output_fd = open(argv[4], O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (output_fd == -1) {
        // error handling
    }
    dup2(output_fd, STDOUT_FILENO);
    close(output_fd);
    
    // Execute the second command
    char **cmd2_args = parse_command(argv[3]);
    execve(cmd2_args[0], cmd2_args, environ);
}

// parent process - cleanup
// Parent closes both ends of the pipe
close(pipe_fd[0]);
close(pipe_fd[1]);

// Wait for both children to finish
waitpid(pid1, NULL, 0);
waitpid(pid2, NULL, 0);
